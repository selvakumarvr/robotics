State Machine

Keypad + RFID Combined Lock

enum State { LOCKED, UNLOCKED, ALARM };
State currentState = LOCKED;
int failCount = 0;

void loop() {
  switch(currentState) {
    case LOCKED:
      checkKeypad();    // check PIN
      checkRFID();      // check card
      break;
    case UNLOCKED:
      runUnlocked();    // servo open, green LED
      break;
    case ALARM:
      runAlarm();       // buzzer + red LED
      break;
  }
}

void grantAccess() {
  currentState = UNLOCKED;
  EEPROM.write(0, EEPROM.read(0) + 1);  // log access count
  failCount = 0;
}

void denyAccess() {
  failCount++;
  if(failCount >= 3) currentState = ALARM;
}
