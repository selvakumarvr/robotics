Row/Column Matrix

4 rows + 4 columns = 8 wires total. Each key connects one row to one column. Arduino detects which pair is connected!

Keypad Library

Handles the scanning automatically. Just define row/col pins and key map — getKey() returns the pressed key!

Code

Reading Keypad Presses

#include <Keypad.h>

const byte ROWS=4, COLS=4;
char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte rowPins[ROWS] = {2,3,4,5};
byte colPins[COLS] = {6,7,8,9};
Keypad kpd = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

void setup() { Serial.begin(9600); }

void loop() {
  char key = kpd.getKey();
  if (key) {
    Serial.println(key);
  }
}
Project

Password Lock System

// Add to above: LED pins + password
const int greenLed=11, redLed=12;
String password = "1234";
String entered = "";

void loop() {
  char key = kpd.getKey();
  if (key) {
    if (key == '#') {         // # = submit
      if (entered == password) {
        digitalWrite(greenLed, HIGH);
        Serial.println("ACCESS GRANTED!");
        delay(3000);
        digitalWrite(greenLed, LOW);
      } else {
        digitalWrite(redLed, HIGH);
        Serial.println("ACCESS DENIED!");
        delay(1000);
        digitalWrite(redLed, LOW);
      }
      entered = "";             // Reset
    } else if (key == '*') { // * = clear
      entered = "";
      Serial.println("Cleared");
    } else {
      entered += key;            // Add digit
      Serial.print("*");       // Show * not digit
    }
  }
}
