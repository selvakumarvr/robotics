Project Plan

How the Robot Works

1

Move forward

Both motors running, robot drives ahead

2

Constantly scan ahead

HC-SR04 measures distance every 100ms

3

Obstacle detected! (<20cm)

Stop motors. Servo scans left and right to measure distances

4

Choose the best direction

Compare left and right distances → pick the wider path

5

Turn and continue

Turn left or right, resume forward motion

Wiring

All Components Connected

📡

HC-SR04

TRIG→Pin 9, ECHO→Pin 10, VCC→5V, GND→GND

🔵

Servo (scan)

Signal→Pin 3 (PWM), VCC→5V, GND→GND. Mount HC-SR04 on top!

⚙️

Motor A (Left)

IN1→Pin 4, IN2→Pin 5, ENA→Pin 6

⚙️

Motor B (Right)

IN3→Pin 7, IN4→Pin 8, ENB→Pin 11

💡 Mount the servo on the front of the robot chassis with the HC-SR04 on the servo arm. This lets the sensor scan left and right without moving the whole robot!

Code — Part 1

Setup & Motor Functions

#include <Servo.h>
Servo scanner;
const int trigPin=9, echoPin=10, servoPin=3;
const int in1=4,in2=5,ena=6,in3=7,in4=8,enb=11;
const int SPEED=180;

float getDistance() {
  digitalWrite(trigPin,LOW); delayMicroseconds(2);
  digitalWrite(trigPin,HIGH); delayMicroseconds(10);
  digitalWrite(trigPin,LOW);
  return pulseIn(echoPin,HIGH) / 58.2;
}
void forward(){ digitalWrite(in1,HIGH);digitalWrite(in2,LOW);analogWrite(ena,SPEED);digitalWrite(in3,HIGH);digitalWrite(in4,LOW);analogWrite(enb,SPEED); }
void backward(){ digitalWrite(in1,LOW);digitalWrite(in2,HIGH);analogWrite(ena,SPEED);digitalWrite(in3,LOW);digitalWrite(in4,HIGH);analogWrite(enb,SPEED); }
void turnLeft(){ analogWrite(ena,0);digitalWrite(in3,HIGH);digitalWrite(in4,LOW);analogWrite(enb,SPEED); }
void turnRight(){ digitalWrite(in1,HIGH);digitalWrite(in2,LOW);analogWrite(ena,SPEED);analogWrite(enb,0); }
void stopAll(){ analogWrite(ena,0);analogWrite(enb,0); }

void setup() {
  pinMode(trigPin,OUTPUT);pinMode(echoPin,INPUT);
  for(int p:{in1,in2,ena,in3,in4,enb}) pinMode(p,OUTPUT);
  scanner.attach(servoPin); scanner.write(90);
  delay(1000);
}
Code — Part 2

Main Loop — Scan & Navigate

void loop() {
  float dist = getDistance();

  if (dist > 20) {
    forward();  // Clear path — go!
  } else {
    stopAll(); delay(300);
    backward(); delay(400);
    stopAll();

    // Scan LEFT
    scanner.write(0); delay(600);
    float leftDist = getDistance();

    // Scan RIGHT
    scanner.write(180); delay(600);
    float rightDist = getDistance();

    // Centre scanner
    scanner.write(90); delay(300);

    // Choose the wider path
    if (leftDist > rightDist) {
      turnLeft(); delay(500);
    } else {
      turnRight(); delay(500);
    }
    stopAll();
  }
  delay(50);
}
